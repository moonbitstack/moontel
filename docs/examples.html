// Copyright 2026 Leo Cheng
// SPDX-License-Identifier: Apache-2.0

///|
/// A tracer, a root span, a child span, some attributes and an event, and the two
/// spans as a sink receives them.
///
///     moon run src/examples/00-hello
#coverage.skip
fn main {
  let recorder = @trace.Recorder::new()
  let tracer = @trace.Tracer::new(
    rand=random_bytes,
    sink=recorder,
    sampler=@trace.parent_based(root=@trace.always),
  )
  // Times are handed in: this library never reads a clock, so a batch job
  // replaying yesterday's events and a server under load use the same code.
  let started = 1_700_000_000_000_000_000L
  let root = tracer.start("GET /orders/{id}", kind=@trace.Server, at=started, attributes=[
    ("http.request.method", @trace.Str("GET")),
  ]) catch {
    _ => abort("the platform has no entropy source")
  }
  let child = tracer.start(
    "SELECT orders",
    parent=root.context(),
    kind=@trace.Client,
    at=started + 1_000_000L,
  ) catch {
    _ => abort("the platform has no entropy source")
  }
  child.set_attribute("db.rows", @trace.Int(3L))
  child.set_status(@trace.Ok)
  child.finish(at=started + 2_000_000L)

  root.add_event("cache.miss", at=started + 3_000_000L)
  root.set_attribute("http.response.status_code", @trace.Int(200L))
  root.finish(at=started + 4_000_000L)

  for span in recorder.spans() {
    println(
      "\{span.name}  \{span.context.to_traceparent()}  \{span.duration()} ns",
    )
    for pair in span.attributes {
      println("  \{pair.0} = \{pair.1}")
    }
    for event in span.events {
      println("  event \{event.name} at \{event.at}")
    }
  }
}

///|
/// Eight or sixteen octets from the platform's entropy source. A caller this close
/// to the edge is the one that knows what to do when there is none; here that is to
/// stop rather than to trace with predictable ids.
fn random_bytes(width : Int32) -> Bytes {
  @moonseed.bytes(width, @system.source) catch {
    _ => abort("the platform has no entropy source")
  }
}

// Copyright 2026 Leo Cheng
// SPDX-License-Identifier: Apache-2.0

///|
/// The two headers a hop reads and the three it writes, over the same carrier: an
/// array of name and value pairs, which is what an ASGI scope and an HTTP client's
/// header list both are.
///
///     moon run src/examples/01-propagate
#coverage.skip
fn main {
  let inbound = [
    ("traceparent", "00-4bf92f3577b34da6a3ce929d0e0e4736-00f067aa0ba902b7-01"),
    ("tracestate", "vendor=abc"),
    ("baggage", "userId=alice,note=caf%C3%A9"),
    ("accept", "application/json"),
  ]
  println("inbound")
  for header in inbound {
    println("  \{header.0}: \{header.1}")
  }

  // What the hop understood: the trace to join, and the bag that came with it.
  let extracted = @propagate.extract(inbound)
  println("")
  println(
    match extracted.context {
      Some(context) => "joined trace \{context.to_traceparent()}"
      None => "no usable context, this hop starts a trace"
    },
  )
  println("baggage: \{extracted.baggage.to_header()}")

  // A span of this hop's own, under whatever context arrived.
  let tracer = @trace.Tracer::new(
    rand=random_bytes,
    sink=@trace.Recorder::new(),
    sampler=@trace.parent_based(root=@trace.always),
  )
  let span = match extracted.context {
    Some(parent) =>
      tracer.start("publish", parent~, kind=@trace.Client, at=1L) catch {
        _ => abort("the platform has no entropy source")
      }
    None =>
      tracer.start("publish", kind=@trace.Client, at=1L) catch {
        _ => abort("the platform has no entropy source")
      }
  }
  span.finish(at=2L)

  // The next request carries this hop's span as its parent, and the bag unchanged.
  let outgoing = @propagate.inject(inbound, {
    context: Some(span.context()),
    baggage: extracted.baggage,
  })
  println("")
  println("outgoing")
  for header in outgoing {
    println("  \{header.0}: \{header.1}")
  }
}

///|
fn random_bytes(width : Int32) -> Bytes {
  @moonseed.bytes(width, @system.source) catch {
    _ => abort("the platform has no entropy source")
  }
}

src/examples/02-asgi/main.mbt moon run src/examples/02-asgi 
// Copyright 2026 Leo Cheng
// SPDX-License-Identifier: Apache-2.0

///|
/// A request through a traced application: the middleware reads the context the
/// request arrived with, opens a server span, and puts the context back on the
/// response.
///
///     moon run src/examples/02-asgi
///
/// The application is driven in process, the way a caller would drive it; a real
/// server binds the same `@spec.AsgiApp` to a socket instead.
#coverage.skip
async fn main {
  @async.pause()
  let recorder = @trace.Recorder::new()
  let tracer = @trace.Tracer::new(
    rand=random_bytes,
    sink=recorder,
    sampler=@trace.parent_based(root=@trace.always),
  )
  let layer = @asgi.Tracing::new(tracer~, now=clock())
  let app : @spec.AsgiApp = async fn(_, receive, send) {
    ignore(receive())
    send(
      @spec.HttpResponseStart(
        status=200,
        headers=[("content-type", "application/json")],
        trailers=false,
      ),
    )
    send(@spec.HttpResponseBody(body=b"{\"ok\":true}", more_body=false))
  }
  let traced = layer.wrap(app)

  let scope = @spec.Http(
    @spec.HttpScope::new(
      http_method="POST",
      path="/orders",
      headers=[
        (
          "traceparent", "00-4bf92f3577b34da6a3ce929d0e0e4736-00f067aa0ba902b7-01",
        ),
        ("baggage", "userId=alice"),
      ],
      server=("shop.example.com", Some(8443)),
    ),
  )

  // A minimal server: one request in, whatever the application sends out.
  let handled = Ref(false)
  let receive : @spec.Receive = async fn() {
    @async.pause()
    if handled.val {
      @spec.HttpDisconnect
    } else {
      handled.val = true
      @spec.HttpRequest(body=b"", more_body=false)
    }
  }
  let send : @spec.Send = async fn(event) {
    @async.pause()
    match event {
      @spec.HttpResponseStart(status~, headers~, ..) => {
        println("response \{status}")
        for header in headers {
          println("  \{header.0}: \{header.1}")
        }
      }
      @spec.HttpResponseBody(body~, ..) =>
        println("body \{body.length()} octets")
      _ => ()
    }
  }
  traced(scope, receive, send)

  println("")
  for span in recorder.spans() {
    println("\{span.name}  \{span.duration()} ns  \{span.status}")
    for pair in span.attributes {
      println("  \{pair.0} = \{pair.1}")
    }
  }
}

///|
/// A clock that moves ten milliseconds per reading. A real service hands in its own
/// clock here; this one keeps the example's output the same on every run.
fn clock() -> () -> Int64 {
  let now = Ref(1_700_000_000_000_000_000L)
  () => {
    now.val = now.val + 10_000_000L
    now.val
  }
}

///|
fn random_bytes(width : Int32) -> Bytes {
  @moonseed.bytes(width, @system.source) catch {
    _ => abort("the platform has no entropy source")
  }
}

///|
suberror Sold {
  Sold(String)
}

///|
#tel.span("orders.place", kind=server, attr="item")
fn place(item : String) -> Int32 raise Sold {
  if item == "gold" {
    raise Sold(item)
  }
  item.length()
}

The function is written once, and says what it wants: a span named orders.place, of kind server, with the item parameter as an attribute. The generator writes the wrapper beside it, and the wrapper is committed — what an annotation means should be readable, not only runnable.

What cmd/span writes, in orders.g.mbt

// Generated by moontel span from orders.mbt. Edit those files, not this one.

///|
fn place_traced(
  tracer : @trace.Tracer,
  item : String,
  now~ : () -> Int64,
  parent? : @trace.TraceContext,
) -> Int32 raise {
  let span = tracer.start(
    "orders.place",
    parent?,
    at=now(),
    kind=@trace.Server,
    attributes=[("item", @trace.Str(item))],
  )
  defer span.finish(at=now())
  let run :() -> Int32 raise = () => place(item)
  match (
    try {
      Ok(run())
    } catch{
      e => Err(e)
    }
  ) {
    Ok(value) => value
    Err(e) => {
      span.record_exception(message=e.to_string(), at=now())
      span.set_status(Error(e.to_string()))
      raise  e
    }
  }
}

place_traced takes the tracer and the clock, because this library has no ambient context to read them from: what the span needs is handed to it. The span finishes whichever way the call ends — a failure is recorded on it with an error status and raised on — and a cancelled span is finished too, which is what the defer is for.

otlp

src/examples/04-otlp/main.mbt moon run src/examples/04-otlp --target native -- http://localhost:4318 
// Copyright 2026 Leo Cheng
// SPDX-License-Identifier: Apache-2.0

///|
/// All three signals leaving the process over OTLP/HTTP: a server span with an
/// event, two log records written inside it, and a counter and a histogram turned
/// into OTLP metrics.
///
///     moon run src/examples/04-otlp --target native
///     moon run src/examples/04-otlp --target native -- http://localhost:4318
///     moon run src/examples/04-otlp --target native -- http://localhost:4318 gzip
///     moon run src/examples/04-otlp --target native -- http://localhost:4318 protobuf
///
/// With an endpoint on the command line the program requires the collector to be
/// there: it exits non-zero when the batch could not be delivered, which is what
/// the conformance job in CI runs. Without one it talks to the protocol's default
/// address and reports what happened, so running the examples one after another
/// stays useful on a machine with no collector.
///
/// `gzip` after the endpoint compresses every request body and says so; `protobuf`
/// sends the binary encoding instead of the JSON one. Either, both or neither, in
/// any order — the conformance job runs all four combinations that differ.
#coverage.skip
async fn main {
  let arguments = @env.args()
  let endpoint = if arguments.length() > 1 {
    arguments[1]
  } else {
    "http://localhost:4318"
  }
  let required = arguments.length() > 1
  // With no collector named there is nothing to retry against, so one attempt says
  // whether something is listening without spending half a minute finding out.
  let attempts = if required { 5 } else { 1 }
  let mut gzip = false
  let mut protobuf = false
  for index in 2..<arguments.length() {
    match arguments[index] {
      "gzip" => gzip = true
      "protobuf" => protobuf = true
      _ => println("ignoring \{arguments[index]}: expected gzip or protobuf")
    }
  }
  let compression = if gzip {
    @http.Compression::Gzip
  } else {
    @http.Compression::Plain
  }
  // The body has two encodings, and the transport is handed the one this run
  // asked for: the JSON mapping, or the binary one the protocol calls default.
  let encoding = if protobuf {
    @binary.protobuf
  } else {
    @otlp.json
  }
  let transport = @http.HttpTransport::new(compression~)
  let started = 1_700_000_000_000_000_000L

  // One exporter per signal. They share the transport and the resource, and each
  // batches and retries on its own.
  let traces = @otlp.TraceExporter::new(
    transport~,
    endpoint~,
    resource=@otlp.Resource::of_service("moontel-example").with_attribute(
      "deployment.environment.name",
      @trace.Str("test"),
    ),
    encoding~,
    now=millis,
    rand=random_bytes,
    attempts~,
  )
  let logs = @otlp.LogsExporter::new(
    transport~,
    endpoint~,
    resource=@otlp.Resource::of_service("moontel-example"),
    encoding~,
    now=millis,
    rand=random_bytes,
    attempts~,
  )
  let metrics = @otlp.MetricsExporter::new(
    transport~,
    endpoint~,
    resource=@otlp.Resource::of_service("moontel-example"),
    encoding~,
    now=millis,
    rand=random_bytes,
    attempts~,
  )

  // A span the sampler keeps, with one child and one event on it.
  let tracer = @trace.Tracer::new(
    sink=traces,
    rand=random_bytes,
    sampler=@trace.parent_based(root=@trace.always),
  )
  let span = tracer.start("GET /orders/42", kind=@trace.Server, at=started, attributes=[
    ("http.request.method", @trace.Str("GET")),
  ]) catch {
    _ => abort("the platform has no entropy source")
  }
  span.add_event("cache.miss", at=started + 1_000_000L)
  span.set_attribute("http.response.status_code", @trace.Int(200L))
  span.set_status(@trace.Ok)

  // The same span is what makes the two log records joinable to the trace.
  let logger = @log.Logger::new(
    sink=logs,
    now=millis,
    scope="moontel-example/orders",
  ).inside(span)
  logger.info(
    "order placed",
    attributes=[("order.id", @trace.Int(42L))],
    at=started + 2_000_000L,
  )
  logger.warn("payment retried once", at=started + 3_000_000L)
  span.finish(at=started + 4_000_000L)

  // A program that already keeps metrics does not keep a second set: this reads the
  // snapshot and the bridge turns it into OTLP. The histogram is registered with its
  // unit, which the exposition appends to the name and the bridge takes back off.
  let registry = @metric.Registry::new()
  let placed = registry.counter("orders_placed", "Orders placed.")
  placed.inc(by=1.0)
  let latency = registry.histogram(
    "order_latency",
    "Order latency.",
    buckets=[0.005, 0.01, 0.025, 0.05, 0.1],
    unit="seconds",
  )
  latency.observe(0.012)
  for metric in @p8s.bridge(registry.collect(at=0.0), at=started) {
    metrics.record(metric)
  }

  let spans = traces.flush()
  let records = logs.flush()
  let points = metrics.flush()
  report("traces", spans)
  report("logs", records)
  report("metrics", points)

  let failed = !delivered(spans) || !delivered(records) || !delivered(points)
  if failed && required {
    abort("a collector was named at \{endpoint} and did not take every signal")
  }
  if failed {
    println(
      "nothing answered at \{endpoint}; start a collector or pass its base URL",
    )
  }
}

///|
/// Whether one flush got its whole batch to the collector.
fn delivered(report : @otlp.Report) -> Bool {
  report.dropped == 0 && report.rejected == 0 && report.attempts > 0
}

///|
fn report(signal : String, outcome : @otlp.Report) -> Unit {
  println(
    "\{signal}: delivered \{outcome.delivered}, rejected \{outcome.rejected}, " +
    "dropped \{outcome.dropped}, requests \{outcome.attempts}",
  )
}

///|
/// The clock the exporters use for batching and for the retry budget, in
/// milliseconds. The runtime's is monotonic, which is what those two need; a
/// wall-clock reading would be needed only to read a `Retry-After` given as a date,
/// and this example hands its records the times it wants instead.
fn millis() -> Int64 {
  @async.now()
}

///|
fn random_bytes(width : Int32) -> Bytes {
  @moonseed.bytes(width, @system.source) catch {
    _ => abort("the platform has no entropy source")
  }
}

The only example that leaves the machine. A tracer hands its span to an exporter, a logger writes two records inside that same span, and a Prometheus snapshot is translated into OTLP metric points — then all three are posted, batched and retried like any other sink.

An endpoint on the command line makes the collector required: the program fails when a signal could not be delivered, which is what the conformance job in CI runs. Without one it talks to the protocol's default address, tries once and says what happened.

// Copyright 2026 Leo Cheng

///|
/// The three signals over OTLP/gRPC: the span, the two log records and the two
/// metrics the HTTP example sends, carried by one unary call each on a channel that
/// stays open between them.
///
///     moon run src/examples/05-grpc --target native
///     moon run src/examples/05-grpc --target native -- localhost:4317
///
/// The payload is the HTTP example's on purpose: a collector that takes both runs
/// holds the same span, the same two lines and the same two metrics twice, once per
/// transport, which is how the conformance job compares the two.
///
/// The argument is `host:port`, because that is what a channel connects to; the
/// protocol puts the gRPC port at 4317. With one on the command line the program
/// requires the collector to be there and exits non-zero when it cannot be reached
/// or when a signal could not be delivered, which is what the conformance job in CI
/// runs.
#coverage.skip
async fn main {
  let arguments = @env.args()
  let address = if arguments.length() > 1 {
    arguments[1]
  } else {
    "127.0.0.1:4317"
  }
  let required = arguments.length() > 1
  let (host, port) = host_and_port(address)
  // A channel connects before anything is queued, so a collector that is not there
  // is known now rather than after a batch was built and lost. The HTTP example
  // learns the same thing at its first request; the report reads the same either
  // way.
  let transport = match channel_to(host, port) {
    Some(transport) => transport
    None => {
      if required {
        abort("a collector was named at \{address} and could not be reached")
      }
      println(
        "nothing answered at \{address}; start a collector on its gRPC port",
      )
      return
    }
  }
  // gRPC carries the binary encoding only, so the exporters are built for it. The
  // endpoint is still a URL: it names the signal, and the transport turns
  // `/v1/traces`, `/v1/logs` and `/v1/metrics` into the methods a collector serves.
  let endpoint = "http://\{address}"
  // With no collector named there is nothing to retry against, so one attempt says
  // whether something is listening without spending half a minute finding out.
  let attempts = if required { 5 } else { 1 }
  let started = 1_700_000_000_000_000_000L
  let traces = @otlp.TraceExporter::new(
    transport~,
    endpoint~,
    resource=@otlp.Resource::of_service("moontel-example").with_attribute(
      "deployment.environment.name",
      @trace.Str("test"),
    ),
    encoding=@binary.protobuf,
    now=millis,
    rand=random_bytes,
    attempts~,
  )
  let logs = @otlp.LogsExporter::new(
    transport~,
    endpoint~,
    resource=@otlp.Resource::of_service("moontel-example"),
    encoding=@binary.protobuf,
    now=millis,
    rand=random_bytes,
    attempts~,
  )
  let metrics = @otlp.MetricsExporter::new(
    transport~,
    endpoint~,
    resource=@otlp.Resource::of_service("moontel-example"),
    encoding=@binary.protobuf,
    now=millis,
    rand=random_bytes,
    attempts~,
  )

  // The span the HTTP example sends, with the same event and the same attributes.
  let tracer = @trace.Tracer::new(
    sink=traces,
    rand=random_bytes,
    sampler=@trace.parent_based(root=@trace.always),
  )
  let span = tracer.start("GET /orders/42", kind=@trace.Server, at=started, attributes=[
    ("http.request.method", @trace.Str("GET")),
  ]) catch {
    _ => abort("the platform has no entropy source")
  }
  span.add_event("cache.miss", at=started + 1_000_000L)
  span.set_attribute("http.response.status_code", @trace.Int(200L))
  span.set_status(@trace.Ok)

  // The same span is what makes the two lines joinable to the trace on this
  // transport too: the ids travel in the record, not in the connection.
  let logger = @log.Logger::new(
    sink=logs,
    now=millis,
    scope="moontel-example/orders",
  ).inside(span)
  logger.info(
    "order placed",
    attributes=[("order.id", @trace.Int(42L))],
    at=started + 2_000_000L,
  )
  logger.warn("payment retried once", at=started + 3_000_000L)
  span.finish(at=started + 4_000_000L)

  // A program that already keeps metrics does not keep a second set: the snapshot
  // is read and the bridge turns it into OTLP, here as everywhere else.
  let registry = @metric.Registry::new()

  placed.inc(by=1.0)
  let latency = registry.histogram(
    "order_latency",
    "Order latency.",
    buckets=[0.005, 0.01, 0.025, 0.05, 0.1],
    unit="seconds",
  )
  latency.observe(0.012)
  for metric in @p8s.bridge(registry.collect(at=0.0), at=started) {
    metrics.record(metric)
  }

  let spans = traces.flush()
  let records = logs.flush()
  let points = metrics.flush()
  transport.close()
  report("traces", spans)
  report("logs", records)
  report("metrics", points)

  let failed = !delivered(spans) || !delivered(records) || !delivered(points)

  }
  if failed {

}

///|
/// Whether one flush got its whole batch to the collector.
fn delivered(report : @otlp.Report) -> Bool {
  report.dropped == 0 && report.rejected == 0 && report.attempts > 0
}

///|
fn report(signal : String, outcome : @otlp.Report) -> Unit {
  println(
    "\{signal}: delivered \{outcome.delivered}, rejected \{outcome.rejected}, " +
    "dropped \{outcome.dropped}, requests \{outcome.attempts}",
  )
}

///|
/// The channel to the collector, or `None` when nothing is listening there: a
/// refused connection is the answer to "is a collector running", not a crash.
async fn channel_to(host : String, port : Int32) -> @grpc.GrpcTransport? {
  Some(@grpc.GrpcTransport::connect(host~, port~)) catch {
    _ => None
  }
}

///|
/// `host:port`, or `host` alone with the protocol's default port.
fn host_and_port(address : String) -> (String, Int32) {
  match address.rev_find(":") {
    Some(at) => (address[:at].to_owned(), port_of(address[at + 1:]))
    None => (address, 4317)
  }
}

///|
/// The port a `host:port` names, or the protocol's 4317 when the text is not a
/// number: a typo should reach the default rather than a port no one serves.
fn port_of(text : StringView) -> Int32 {
  let mut port = 0
  for digit in text {
    if digit < '0' || digit > '9' {
      return 4317
    }
    port = port * 10 + (digit.to_int() - '0'.to_int())
  }
  if port == 0 {
    4317
  } else {
    port
  }
}

///|
/// The clock the exporters use for batching and for the retry budget. The runtime's
/// is monotonic, which is what those two need; the times the signals carry are the
/// fixed ones above, in nanoseconds.
fn millis() -> Int64 {
  @async.now()
}

///|
/// Sixteen bytes from the platform's entropy source, for a span id and a trace id.
fn random_bytes(width : Int32) -> Bytes {
  @moonseed.bytes(width, @system.source) catch {
    _ => abort("the platform has no entropy source")
  }
}

The three signals the HTTP example sends, over the other transport the protocol defines: one span with an event, two log records written inside it, and a counter and a histogram out of a registry's snapshot. gRPC carries the binary protobuf, so each exporter is built for it; everything else — the queue, the batch, the retry rules and the two size bounds — is the code the HTTP path runs.

The payload is deliberately the HTTP example's: a collector that takes both runs holds the same span, the same two lines and the same two metrics twice, once per transport. That is what the conformance job in CI counts, which is how “the same item on every protocol” is checked rather than asserted.

The argument is host:port, because that is what a channel connects to, and the protocol puts the gRPC port at 4317. The endpoint handed to the exporter is still a URL: it names the signal, and the transport turns /v1/traces, /v1/logs and /v1/metrics into the methods a collector serves.

///|
/// Eight instruments, three views, one exporter.
///
///     moon run src/examples/06-meter --target native -- http://localhost:4318
///     moon run src/examples/06-meter --target native -- dump
///
/// With an endpoint the program requires the collector to be there. With `dump`
/// it writes one `ExportMetricsServiceRequest` to standard output, which is the
/// payload the live-check job reads back and compares against the semantic
/// conventions. Both paths come from the same `MeterProvider::collect`.
///
/// The point of the picture is what the meter does that a plain counter does not:
/// three instruments report at collection time, one distribution adapts its buckets
/// to the measurements, and three views say what the metrics look like on the wire —
/// a name, a set of attributes, and an instrument nobody asked for.
#coverage.skip
async fn main {
  let arguments = @env.args()
  let mode = if arguments.length() > 1 {
    arguments[1]
  } else {
    "http://localhost:4318"
  }
  let started = 1_700_000_000_000_000_000L
  let resource = @otlp.Resource::of_service("moontel-example")
  // `openai` is one of the provider names the semantic conventions enumerate; a
  // name outside that list is a value no dashboard can join on.
  let attributes = @meter.gen_ai_client_attributes(
    provider="openai",
    operation="chat",
    request_model="gpt-4o",
    response_model="gpt-4o-2024-08-06",
  )
  if mode == "dump" {
    let recorder = @otlp.MetricRecorder::new()
    let provider = @meter.MeterProvider::new(
      sink=recorder,
      now=() => started,
      views=views(),
      current=() => Some(request_span),
      start=started,
    )
    record(provider, attributes, started) catch {
      _ => abort("an instrument name was refused")
    }
    println(
      @otlp.encode_metrics(provider.collect(at=started + 1_000_000L), resource~),
    )
    return
  }
  let transport = @http.HttpTransport::new()
  let exporter = @otlp.MetricsExporter::new(
    transport~,
    endpoint=mode,
    resource~,
    now=millis,
    rand=random_bytes,
  )
  let provider = @meter.MeterProvider::new(
    sink=exporter,
    now=millis,
    views=views(),
    current=() => Some(request_span),
    start=started,
  )
  record(provider, attributes, started) catch {
    _ => abort("an instrument name was refused")
  }
  // `emit` queues the metrics on the same exporter that carries spans and logs;
  // `flush` is what puts them on the wire.
  provider.emit(at=started + 1_000_000L) |> ignore
  let report = exporter.flush()
  println(
    "metrics: delivered \{report.delivered}, rejected \{report.rejected}, " +
    "dropped \{report.dropped}, requests \{report.attempts}",
  )
  if report.dropped != 0 || report.rejected != 0 || report.attempts == 0 {
    abort("a collector was named at \{mode} and did not take every metric")
  }
}

///|
/// The span the program is inside while it records. A real service hands the span it
/// already has; here it is fixed, so the exemplar a backend shows names the same
/// trace on every run.
let request_span : @trace.TraceContext = @trace.TraceContext::new(
  trace_id=b"\x4b\xf9\x2f\x35\x77\xb3\x4d\xa6\xa3\xce\x92\x9d\x0e\x0e\x47\x36",
  span_id=b"\x00\xf0\x67\xaa\x0b\xa9\x02\xb7",
  flags=@trace.sampled_flag,
)

///|
/// What the three views say about the metrics this program registers.
///
/// A view is the only place a decision about the wire belongs: the call sites keep
/// writing the measurements they can measure, and the program decides here that a
/// token count does not need the response model, that a queue depth reads better as a
/// length, and that a debug tick is not worth sending at all.
fn views() -> Array[@meter.View] {
  [
    @meter.View::new(
      instrument="gen_ai.client.token.usage",
      attribute_keys=[
        @semconv.gen_ai_provider_name, @semconv.gen_ai_operation_name,
        @semconv.gen_ai_request_model,
      ],
    ),
    @meter.View::new(
      instrument="gen_ai.client.queue.depth",
      name="gen_ai.client.queue.length",
      attribute_keys=[
        @semconv.gen_ai_provider_name, @semconv.gen_ai_operation_name,
      ],
    ),
    @meter.View::new(instrument="moontel.example.debug.tick", aggregation=@meter.Drop),
  ]
}

///|
/// The instruments a service would register once and write to on every request.
/// Every name and every attribute key keeps its dots, because OTLP and the semantic
/// conventions both spell them that way.
fn record(
  provider : @meter.MeterProvider,
  attributes : @meter.Attributes,
  started : Int64,
) -> Unit raise @meter.Refused {
  let meter = provider.meter(name="moontel-example/orders")
  let tokens = meter.counter(
    "gen_ai.client.token.usage",
    description="Tokens used by the client.",
    unit="{token}",
  )
  tokens.add(128.0, attributes~, at=started, context=request_span)
  let duration = meter.histogram(
    @semconv.gen_ai_client_operation_duration,
    description="GenAI operation duration.",
    unit="s",
  )
  duration.record(6.0, attributes~, at=started, context=request_span)
  let active = meter.up_down_counter(
    "gen_ai.client.active.requests",
    description="Requests in flight.",
    unit="{request}",
  )
  active.add(3.0, attributes~, at=started)
  active.add(-1.0, attributes~, at=started)
  // An observed gauge: the callback runs at each collection, so the number is what
  // the queue holds now rather than what it held when a request last ran.
  let _ = meter.observable_gauge(
    "gen_ai.client.queue.depth",
    description="Requests waiting.",
    unit="{request}",
    observe=fn(observer) { observer.observe(7.0, attributes~, at=started) },
  )
  let _ = meter.observable_counter(
    "gen_ai.client.tokens.reported",
    description="Tokens the client has used, as the provider counts them.",
    unit="{token}",
    observe=fn(observer) { observer.observe(512.0, attributes~, at=started) },
  )
  let _ = meter.observable_up_down_counter(
    "gen_ai.client.workers.reported",
    description="Workers the client keeps for the model.",
    unit="{worker}",
    observe=fn(observer) { observer.observe(4.0, attributes~, at=started) },
  )
  // A distribution whose buckets follow the measurements: the five values below span
  // four orders of magnitude, which explicit boundaries at the wrong scale would
  // throw into one bucket.
  let latency = meter.exponential_histogram(
    "moontel.example.latency",
    description="Time a request took, at any scale.",
    unit="s",
  )
  for value in [0.004, 0.011, 0.052, 0.19, 1.7] {
    latency.record(value, attributes~, at=started, context=request_span)
  }
  // Registered, written, and dropped by a view: the point of a view is that the call
  // site does not have to know.
  let tick = meter.counter(
    "moontel.example.debug.tick",
    description="A tick nobody needs on a dashboard.",
  )
  tick.inc(at=started)
}

///|
fn millis() -> Int64 {
  @async.now()
}

///|
fn random_bytes(width : Int32) -> Bytes {
  @moonseed.bytes(width, @system.source) catch {
    _ => abort("the platform has no entropy source")
  }
}

Eight instruments, one attribute group and one exporter. The counter and the up-down counter are sums on the wire, differing in whether a measurement may go down; the histogram fills the fifteen explicit boundaries the metrics SDK specifies. Every name and every attribute key keeps its dots, because that is the spelling a backend and a dashboard both match on.

Three of the eight report themselves at collection time: the gauge states the queue depth when the reader asks, and a delta reader gets the difference rather than the total. One distribution is exponential, so its buckets follow the values instead of a boundary list chosen in advance.

Three views decide what reaches the wire without a call site changing: the token counter loses an attribute, the queue depth is renamed, and a debug tick is registered and dropped. Every point carries an exemplar, which is the trace of the request it belongs to.

dump writes the same payload to standard output instead of sending it. The live-check job reads that body, flattens it into the sample list Weaver ingests, and compares the seven metrics, their units and their attributes against a registry; the same payload is posted to a real collector, which is where the names, units, attributes, data types and exemplars are counted back out of its log.

Run them yourself.

The seven programs are packages in this repository, so a checkout can run them without installing anything else: 

moon run src/examples/00-hello --target native
moon run src/examples/01-propagate --target native
moon run src/examples/02-asgi --target native
moon run src/examples/04-otlp --target native -- http://localhost:4318
moon run src/examples/05-grpc --target native -- 127.0.0.1:4317
moon run src/examples/06-meter --target native -- http://localhost:4318

00-hello and 01-propagate build on every target. 02-asgi awaits what the application sends, and the async runtime has no wasm-gc build, so that target builds a second file that says so rather than a program that only looks like the example. 04-otlp needs a socket, which the JavaScript backends do not have, so it is built for native and wasm only; 05-grpc needs a channel, which is a TCP connection, so it is native alone; 06-meter uses the HTTP transport and builds like 04-otlp.

The home page runs a fourth program: it compiles this library to JavaScript and draws what came out. That one is not meant to be run by hand: the page loads it as a module and draws its output.
